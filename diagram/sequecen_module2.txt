activate Home

Home -> Selection : call
activate Selection

Revenue --> Manager : display date range form

Revenue -> Revenue : validate date range
activate Revenue
deactivate Revenue

activate RevenueDAO

activate CustomerStatistics
CustomerStatistics -> CustomerStatistics : CustomerRevenueStatistics4()
activate CustomerStatistics
deactivate CustomerStatistics

deactivate CustomerStatistics

deactivate RevenueDAO

deactivate RevenueDAO

Revenue --> Manager : display customers and revenue (descending)\nwith a sort list when results exist

autonumber stop
loop Manager changes the revenue sort order
  Manager -> Revenue : select ascending or descending revenue
  Revenue -> Revenue : sort loaded results and update ranks
  Revenue --> Manager : display reordered results
end

Manager -> Revenue : select a customer
Revenue -> History : call: customerId, startDate, endDate
activate History

History -> OrderDAO : call

OrderDAO -> OrderDAO : getOrdersByCustomer(customerId, startDate, endDate)

OrderDAO -> Order : call
activate Order
Order -> Order : SalesOrder4()
activate Order
deactivate Order
Order --> OrderDAO : return

activate History
deactivate History

Manager -> History : select an order
History -> OrderView : call: orderId, customerId, startDate, endDate
activate OrderView

OrderView -> OrderView : get order
activate OrderView
deactivate OrderView

OrderView -> DetailDAO : call
activate DetailDAO
DetailDAO -> DetailDAO : getOrderDetails(orderId)
activate DetailDAO

DetailDAO -> OrderDetail : call
activate OrderDetail
OrderDetail -> OrderDetail : SalesOrderDetail4()
activate OrderDetail
deactivate OrderDetail
OrderDetail --> DetailDAO : return
deactivate OrderDetail

deactivate DetailDAO
DetailDAO --> OrderView : return: SalesOrderDetail4[]
deactivate DetailDAO

OrderView --> Manager : display order information, product list and total amount

deactivate Home

@startuml
skinparam Style strictuml
skinparam SequenceMessageAlignment center
skinparam monochrome true
skinparam lifelineStrategy solid
hide footbox
autonumber 1 "0:"

actor "Customer" as Customer
boundary "CustomerHomeView4.jsp" as Home
boundary "ProductSearchView4.jsp" as Search
boundary "ProductDetailView4.jsp" as Detail
control "ProductDAO4" as PDAO
entity "Product4" as Product

activate Customer
activate Home

Customer -> Home : select Product Search
Home -> Search : call
activate Search

Search -> PDAO : call
activate PDAO
PDAO -> PDAO : getProductsByKeyword("")
activate PDAO

PDAO -> Product : call
activate Product
Product -> Product : Product4()
activate Product
deactivate Product
Product --> PDAO : return
deactivate Product

deactivate PDAO
PDAO --> Search : return: Product4[]
deactivate PDAO

Search --> Customer : display search form

Customer -> Search : enter keyword and click Search
Search -> Search : trim and validate keyword
activate Search
deactivate Search

Search -> PDAO : call
activate PDAO
PDAO -> PDAO : getProductsByKeyword(keyword)
activate PDAO

PDAO -> Product : call
activate Product
Product -> Product : Product4()
activate Product
deactivate Product
Product --> PDAO : return
deactivate Product

deactivate PDAO
PDAO --> Search : return: Product4[]
deactivate PDAO

Search --> Customer : display product list

Customer -> Search : select a product
Search -> Detail : call
activate Detail

Detail -> Detail : receive productId
activate Detail
deactivate Detail

Detail -> PDAO : call
activate PDAO
PDAO -> PDAO : getProductById(productId)
activate PDAO

PDAO -> Product : call
activate Product
Product -> Product : Product4()
activate Product
deactivate Product
Product --> PDAO : return
deactivate Product

deactivate PDAO
PDAO --> Detail : return: Product4
deactivate PDAO

Detail --> Customer : display product details

deactivate Detail
deactivate Search
deactivate Home
deactivate Customer
@enduml
